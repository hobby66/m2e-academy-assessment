-- 1. Create Topics Table CREATE TABLE IF NOT EXISTS public.topics ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), name text NOT NULL, is_active boolean DEFAULT false, admin_id uuid DEFAULT auth.uid() ); -- 2. Add topic_id and admin_id to existing tables ALTER TABLE public.questions ADD COLUMN IF NOT EXISTS topic_id uuid REFERENCES public.topics(id); ALTER TABLE public.questions ADD COLUMN IF NOT EXISTS admin_id uuid DEFAULT auth.uid(); ALTER TABLE public.submissions ADD COLUMN IF NOT EXISTS topic_id uuid REFERENCES public.topics(id); -- 3. Migrate existing data to a default topic DO $$ DECLARE default_topic_id uuid; default_admin_id uuid; BEGIN -- Get the first admin user from the auth table SELECT id INTO default_admin_id FROM auth.users LIMIT 1; -- Create a default topic if we have an admin IF default_admin_id IS NOT NULL THEN INSERT INTO public.topics (name, is_active, admin_id) VALUES ('Default Topic (Migrated)', true, default_admin_id) RETURNING id INTO default_topic_id; -- Assign existing questions and submissions to this topic UPDATE public.questions SET topic_id = default_topic_id, admin_id = default_admin_id WHERE topic_id IS NULL; UPDATE public.submissions SET topic_id = default_topic_id WHERE topic_id IS NULL; END IF; END $$; -- 4. Enable RLS on Topics ALTER TABLE public.topics ENABLE ROW LEVEL SECURITY; -- 5. RLS Policies for Topics DROP POLICY IF EXISTS "Admins can manage their own topics" ON public.topics; CREATE POLICY "Admins can manage their own topics" ON public.topics FOR ALL TO authenticated USING (admin_id = auth.uid()); DROP POLICY IF EXISTS "Public can view active topics" ON public.topics; CREATE POLICY "Public can view active topics" ON public.topics FOR SELECT TO anon USING (is_active = true); -- 6. Update RLS for Questions & Submissions (Strict Multi-Tenancy) DROP POLICY IF EXISTS "Admin full access to questions" ON public.questions; CREATE POLICY "Admin full access to questions" ON public.questions FOR ALL TO authenticated USING (admin_id = auth.uid()); DROP POLICY IF EXISTS "Admin can read submissions" ON public.submissions; CREATE POLICY "Admin can read submissions" ON public.submissions FOR SELECT TO authenticated USING (topic_id IN (SELECT id FROM public.topics WHERE admin_id = auth.uid())); DROP POLICY IF EXISTS "Admin can delete submissions" ON public.submissions; CREATE POLICY "Admin can delete submissions" ON public.submissions FOR DELETE TO authenticated USING (topic_id IN (SELECT id FROM public.topics WHERE admin_id = auth.uid())); -- 7. Update RPC: Get Questions securely based on Topic CREATE OR REPLACE FUNCTION get_questions_for_participant(p_topic_id uuid) RETURNS json AS $$ SELECT json_agg(json_build_object('id', id, 'text', text, 'options', options)) FROM questions WHERE topic_id = p_topic_id; $$ LANGUAGE sql SECURITY DEFINER; -- 8. Update RPC: Submit Assessment securely (Handles Option Shuffling) CREATE OR REPLACE FUNCTION submit_assessment( p_topic_id uuid, p_full_name text, p_region text, p_branch text, p_email text, p_pf_number text, p_mode text, p_answers jsonb ) RETURNS json AS $$ DECLARE v_score int := 0; v_total int := 0; v_correct_count int := 0; v_detail jsonb := '[]'::jsonb; v_ans record; v_q record; v_is_correct boolean; BEGIN v_total := jsonb_array_length(p_answers); FOR v_ans IN SELECT * FROM jsonb_to_recordset(p_answers) AS x("questionId" text, "selectedText" text) LOOP SELECT text, options, correct_index INTO v_q FROM questions WHERE id = v_ans."questionId"; IF FOUND THEN -- Compare by TEXT (allows option shuffling on frontend) v_is_correct := (v_q.options->>v_q.correct_index = v_ans."selectedText"); IF v_is_correct THEN v_correct_count := v_correct_count + 1; END IF; v_detail := v_detail || jsonb_build_object( 'questionId', v_ans."questionId", 'question', v_q.text, 'options', v_q.options, 'selectedText', v_ans."selectedText", 'correctText', v_q.options->>v_q.correct_index, 'isCorrect', v_is_correct ); END IF; END LOOP; IF v_total > 0 THEN v_score := round((v_correct_count::numeric / v_total::numeric) * 100); END IF; INSERT INTO submissions (topic_id, full_name, region, branch, email, pf_number, mode, score, correct_count, total_questions, answers) VALUES (p_topic_id, p_full_name, p_region, p_branch, p_email, p_pf_number, p_mode, v_score, v_correct_count, v_total, v_detail); RETURN json_build_object('score', v_score, 'correct_count', v_correct_count, 'total', v_total, 'answers', v_detail); END; $$ LANGUAGE plpgsql SECURITY DEFINER;
